public.tenants (
  id UUID PRIMARY KEY,

  subdomain VARCHAR(255) UNIQUE,
  stripe_customer_id VARCHAR(255),
  subscription_status VARCHAR(50),
  plan_type VARCHAR(50),
  created_at TIMESTAMP,

  id UUID PRIMARY KEY,
  email VARCHAR(255) UNIQUE,
  password_hash TEXT,
  tenant_id UUID REFERENCES tenants(id),
  role VARCHAR(50), -- 'super_admin', 'company_admin', 'user'
  is_active BOOLEAN DEFAULT true,
  created_at TIMESTAMP,
  updated_at TIMESTAMP
)

tenant_data.customers (
  id UUID PRIMARY KEY,
  tenant_id UUID REFERENCES tenants(id),
  first_name VARCHAR(100),
  last_name VARCHAR(100),
  email VARCHAR(255),
  phone VARCHAR(50),
  company VARCHAR(255),
  status VARCHAR(50), -- 'lead', 'customer', 'prospect'
  created_at TIMESTAMP,
  updated_at TIMESTAMP

tenant_data.leads (
  id UUID PRIMARY KEY,
  tenant_id UUID REFERENCES tenants(id),

  value DECIMAL(10,2),
  assigned_to UUID REFERENCES users(id),
  created_at TIMESTAMP,
  updated_at TIMESTAMP

tenant_data.deals (
  id UUID PRIMARY KEY,
  tenant_id UUID REFERENCES tenants(id),
  lead_id UUID REFERENCES leads(id),
  stage VARCHAR(50), -- 'prospecting', 'qualification', 'proposal', 'negotiation', 'closed_won', 'closed_lost'
  value DECIMAL(10,2),
  probability INTEGER, -- 0-100 percentage
  close_date DATE,
  assigned_to UUID REFERENCES users(id),
  created_at TIMESTAMP,
  updated_at TIMESTAMP
)

tenant_data.tasks (
  id UUID PRIMARY KEY,
  tenant_id UUID REFERENCES tenants(id),
  title VARCHAR(255),
  description TEXT,
  assigned_to UUID REFERENCES users(id),
  due_date TIMESTAMP,
  status VARCHAR(50), -- 'pending', 'in_progress', 'completed'
  priority VARCHAR(20), -- 'low', 'medium', 'high'
  created_by UUID REFERENCES users(id),
  created_at TIMESTAMP,
  updated_at TIMESTAMP
)

-- Row-level security policies
CREATE POLICY tenant_isolation_policy ON tenant_data.customers
  FOR ALL TO authenticated_user
  USING (tenant_id = current_setting('app.current_tenant')::uuid);

KEY MIDDLEWARE COMPONENTS
------------------------
1. Tenant Resolver Middleware
   - Extracts tenant info from subdomain or header
   - Sets current tenant in request context
   - Validates tenant existence and subscription status

2. Authentication Middleware
   - Verifies JWT tokens
   - Attaches user info to request object
   - Checks user status and permissions

3. Authorization Middleware
   - Validates user permissions for resources
   - Implements role-based access control
   - Prevents unauthorized access to tenant data

MULTI-TENANCY IMPLEMENTATION
---------------------------
Method 1: Single Database with Row-Level Security (Recommended)
- All tenants share one database
- Tenant isolation via tenant_id column and RLS
- Cost-effective and easy to maintain
- Requires careful attention to security

Method 2: Separate Schemas per Tenant
- Each tenant gets their own schema
- Stronger isolation than RLS
- More complex management
- Better performance isolation

// Tenant resolver middleware
export const resolveTenant = async (req: Request, res: Response, next: NextFunction) => {
  // Extract tenant from subdomain or header
  const host = req.get('Host');
  const subdomain = host?.split('.')[0]; // e.g., company1.yourcrm.com
  
  if (!subdomain) {
    return res.status(400).json({ error: 'Invalid tenant identifier' });
  }
  
  // Find tenant by subdomain
  const tenant = await prisma.tenant.findUnique({
    where: { subdomain }
  });
  
  if (!tenant) {
    return res.status(404).json({ error: 'Tenant not found' });
  }
  
  // Check subscription status
  if (tenant.subscription_status !== 'active') {
    return res.status(402).json({ error: 'Subscription inactive' });
  }
  
  // Attach tenant to request
  req.tenant = tenant;
  
  // Set tenant context for database queries
  await prisma.$executeRaw`SELECT set_config('app.current_tenant', ${tenant.id}, false)`;
  
  next();
};

// Authentication middleware
export const authenticate = async (req: Request, res: Response, next: NextFunction) => {
  const token = req.header('Authorization')?.replace('Bearer ', '');
  
  if (!token) {
    return res.status(401).json({ error: 'Access denied. No token provided.' });
  }
  
  try {
    const decoded = jwt.verify(token, process.env.JWT_SECRET!) as JwtPayload;
    
    // Get user with tenant association
    const user = await prisma.user.findUnique({
      where: { id: decoded.userId },
      include: { tenant: true }
    });
    
    if (!user || !user.isActive) {
      return res.status(401).json({ error: 'Invalid token or user inactive.' });
    }
    
    // Verify user belongs to current tenant (for non-super admins)
    if (user.role !== 'super_admin' && user.tenantId !== req.tenant?.id) {
      return res.status(403).json({ error: 'Access forbidden' });
    }
    
    req.user = user;
    next();
  } catch (error) {
    res.status(400).json({ error: 'Invalid token.' });
  }
};

// Example controller with tenant isolation
export const getCustomers = async (req: Request, res: Response) => {
  try {
    const customers = await prisma.customer.findMany({
      where: {
        tenantId: req.tenant!.id  // Always filter by current tenant
      },
      orderBy: { createdAt: 'desc' }
    });
    
    res.json(customers);
  } catch (error) {
    res.status(500).json({ error: 'Failed to fetch customers' });
  }
};
