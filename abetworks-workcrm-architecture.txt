- More complex management for ABETWORKS
- Better performance isolation

// Tenant resolver middleware for ABETWORKS WORKCRM
export const resolveTenant = async (req: Request, res: Response, next: NextFunction) => {
  // Extract tenant from subdomain or header for ABETWORKS WORKCRM
  const host = req.get('Host');
  const subdomain = host?.split('.')[0]; // e.g., client1.workcrm.abetworks.com
  
  if (!subdomain) {
    return res.status(400).json({ error: 'Invalid tenant identifier for ABETWORKS WORKCRM' });
  }
  
  // Find tenant by subdomain for ABETWORKS platform
  const tenant = await prisma.tenant.findUnique({
    where: { subdomain }
  });
  
  if (!tenant) {
    return res.status(404).json({ error: 'Tenant not found in ABETWORKS WORKCRM' });
  }
  
  // Check subscription status for ABETWORKS platform
  if (tenant.subscription_status !== 'active') {
    return res.status(402).json({ error: 'ABETWORKS WORKCRM subscription inactive' });
  }
  
  // Attach tenant to request for ABETWORKS WORKCRM
  req.tenant = tenant;
  
  // Set tenant context for database queries in ABETWORKS platform
  await prisma.$executeRaw`SELECT set_config('app.current_tenant', ${tenant.id}, false)`;
  
  next();
};

// Authentication middleware for ABETWORKS WORKCRM
export const authenticate = async (req: Request, res: Response, next: NextFunction) => {
  const token = req.header('Authorization')?.replace('Bearer ', '');
  
  if (!token) {
    return res.status(401).json({ error: 'Access denied. No token provided for ABETWORKS WORKCRM.' });
  }
  
  try {
    const decoded = jwt.verify(token, process.env.JWT_SECRET!) as JwtPayload;
    
    // Get user with tenant association for ABETWORKS WORKCRM
    const user = await prisma.user.findUnique({
      where: { id: decoded.userId },
      include: { tenant: true }
    });
    
    if (!user || !user.isActive) {
      return res.status(401).json({ error: 'Invalid token or user inactive in ABETWORKS WORKCRM.' });
    }
    
    // Verify user belongs to current tenant (for non-super admins) in ABETWORKS platform
    if (user.role !== 'abetworks_super_admin' && user.tenantId !== req.tenant?.id) {
      return res.status(403).json({ error: 'Access forbidden in ABETWORKS WORKCRM' });
    }
    
    req.user = user;
    next();
  } catch (error) {
    res.status(400).json({ error: 'Invalid token for ABETWORKS WORKCRM.' });
  }
};

// Example controller with tenant isolation for ABETWORKS WORKCRM
export const getCustomers = async (req: Request, res: Response) => {
  try {
    const customers = await prisma.customer.findMany({
      where: {
        tenantId: req.tenant!.id  // Always filter by current tenant for ABETWORKS WORKCRM
      },
      orderBy: { createdAt: 'desc' }
    });
    
    res.json(customers);
  } catch (error) {
    res.status(500).json({ error: 'Failed to fetch customers in ABETWORKS WORKCRM' });
  }
};
